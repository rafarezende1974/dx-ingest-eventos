name: Validate and build Windows v2

on:
  push:
    branches: [main, 'codex/editor-assistente-v2']
  pull_request:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
          cache: pip
      - name: Install pinned dependencies
        run: python -m pip install -r requirements-build.txt
      - name: Regression and interface tests
        run: python -m unittest discover -s tests -v
      - name: Generate assets
        run: python generate_assets.py
      - name: Build application folder with theme data
        run: python -m PyInstaller --noconfirm --clean --onedir --windowed --name EditorAssistente --icon assets/icon.ico --collect-all customtkinter --add-data "assets;assets" app.py
      - name: Smoke test packaged application
        shell: pwsh
        run: |
          $app = Start-Process -FilePath dist/EditorAssistente/EditorAssistente.exe -ArgumentList '--smoke-test' -PassThru -WindowStyle Hidden
          if (-not $app.WaitForExit(30000)) { $app.Kill(); throw 'Application startup timed out' }
          if ($app.ExitCode -ne 0) { throw "Application smoke test failed: $($app.ExitCode)" }
      - uses: actions/upload-artifact@v4
        with:
          name: editor-assistente-v2-preview
          path: dist/EditorAssistente/
          if-no-files-found: error
