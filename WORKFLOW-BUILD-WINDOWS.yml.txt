name: Validate and build Windows v2

on:
  push:
    branches: [main, 'codex/editor-assistente-v2']
    tags: ['v2.0.0']
  pull_request:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
          cache: pip
      - name: Install pinned dependencies
        run: python -m pip install -r requirements-build.txt
      - name: Regression and interface tests
        run: python -m unittest discover -s tests -v
      - name: Generate assets
        run: python generate_assets.py
      - name: Build application folder with theme data
        run: python -m PyInstaller --noconfirm --clean --onedir --windowed --name EditorAssistente --icon assets/icon.ico --collect-all customtkinter --add-data "assets;assets" app.py
      - name: Smoke test packaged application
        shell: pwsh
        run: |
          $app = Start-Process -FilePath dist/EditorAssistente/EditorAssistente.exe -ArgumentList '--smoke-test' -PassThru -WindowStyle Hidden
          if (-not $app.WaitForExit(30000)) { $app.Kill(); throw 'Application startup timed out' }
          if ($app.ExitCode -ne 0) { throw "Application smoke test failed: $($app.ExitCode)" }
      - uses: actions/upload-artifact@v4
        with:
          name: editor-assistente-2.0.0-windows
          path: dist/EditorAssistente/
          if-no-files-found: error

  release:
    if: github.event_name == 'push' && github.ref == 'refs/tags/v2.0.0'
    needs: build
    runs-on: windows-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - name: Verify release belongs to main
        shell: pwsh
        run: |
          git merge-base --is-ancestor HEAD origin/main
          if ($LASTEXITCODE -ne 0) { throw 'Release commit must be on main' }
      - uses: actions/download-artifact@v4
        with:
          name: editor-assistente-2.0.0-windows
          path: package/EditorAssistente
      - name: Package and publish official version
        shell: pwsh
        env:
          GH_TOKEN: ${{ github.token }}
          GH_REPO: ${{ github.repository }}
        run: |
          Copy-Item README.md package/EditorAssistente/README.md
          Copy-Item RELEASE-NOTES.md package/EditorAssistente/RELEASE-NOTES.md
          Compress-Archive -Path package/EditorAssistente -DestinationPath EditorAssistente-2.0.0-Windows.zip
          $hash = (Get-FileHash EditorAssistente-2.0.0-Windows.zip -Algorithm SHA256).Hash.ToLower()
          "$hash  EditorAssistente-2.0.0-Windows.zip" | Set-Content SHA256SUMS.txt -Encoding ascii
          gh release create v2.0.0 EditorAssistente-2.0.0-Windows.zip SHA256SUMS.txt --verify-tag --title 'Editor Assistente 2.0.0' --notes-file RELEASE-NOTES.md
          if ($LASTEXITCODE -ne 0) { throw 'Release publication failed' }
